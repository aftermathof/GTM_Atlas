\newcommand{\resumesection}[2]{%
  \section{\makebox[\resumeiconwidth][c]{\color{primary}#1}\hspace{0.35em}#2}%
}
\newcommand{\resumeentry}[3]{%
  \textbf{#1}\enspace\textbar\enspace#2\hfill{#3}\par
}
\newcommand{\resumeeducation}[4]{%
  \textbf{#1}\enspace#2\hfill{#3}\par
  \begin{itemize}\item #4\end{itemize}
}

% ---------- 项目经历（可选、可重复） ----------
\resumesection{\faProjectDiagram}{项目经历}
\resumeentry{项目名称}{项目角色}{起止年月}
\begin{itemize}
  \item \textbf{项目简介：}简述业务背景、职责、实施措施和结果。
  \item \textbf{项目成果：}填写可验证的业务影响或量化指标。
\end{itemize}

% ---------- 实习或工作经历（可重复） ----------
\resumesection{\faBriefcase}{实习经历}
\resumeentry{公司名称}{职位名称}{起止年月}
\begin{itemize}
  \item \textbf{工作内容：}说明职责、采取的行动和使用的方法。
  \item \textbf{成果产出：}说明业务价值；无数据时不要虚构量化结果。
\end{itemize}

% ---------- 技能与语言（可选） ----------
\resumesection{\faTools}{技能特长}
\begin{itemize}
  \item \textbf{技能类别：}工具、技术或业务能力。
  \item \textbf{语言能力：}考试或实际工作语言水平。
\end{itemize}
